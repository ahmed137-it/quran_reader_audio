class SurahReaderState {
  final SurahDetailModel surah;
  final List<QuranPageModel> pages;
  final int currentPage;

  const SurahReaderState({
    required this.surah,
    required this.pages,
    required this.currentPage,
  });

  QuranPageModel get page => pages[currentPage];

  bool get canGoNext => currentPage < pages.length - 1;
  bool get canGoPrevious => currentPage > 0;

  SurahReaderState copyWith({
    int? currentPage,
  }) {
    return SurahReaderState(
      surah: surah,
      pages: pages,
      currentPage: currentPage ?? this.currentPage,
    );
  }
}

class SurahReaderCubit extends Cubit<SurahReaderState> {
  SurahReaderCubit(SurahDetailModel surah)
      : super(
    SurahReaderState(
      surah: surah,
      pages: SurahPaginationHelper.paginateAyahs(surah.ayahs),
      currentPage: 0,
    ),
  );

  void nextPage() {
    if (state.canGoNext) {
      emit(state.copyWith(currentPage: state.currentPage + 1));
    }
  }

  void previousPage() {
    if (state.canGoPrevious) {
      emit(state.copyWith(currentPage: state.currentPage - 1));
    }
  }

  void goToPage(int page) {
    if (page >= 0 && page < state.pages.length) {
      emit(state.copyWith(currentPage: page));
    }
  }
}

class SurahDetailModel {
  final int surahNo;
  final String surahName;
  final String surahNameArabic;
  final String surahNameArabicLong;
  final String surahNameTranslation;
  final String revelationPlace;
  final int totalAyah;
  final Map<String, AudioInfo> audio;
  final List<AyahModel> ayahs;

  const SurahDetailModel({
    required this.surahNo,
    required this.surahName,
    required this.surahNameArabic,
    required this.surahNameArabicLong,
    required this.surahNameTranslation,
    required this.revelationPlace,
    required this.totalAyah,
    required this.audio,
    required this.ayahs,
  });

  bool get isMakki => revelationPlace.toLowerCase().startsWith('mecc');

  bool get hasBismillahHeader => surahNo != 1 && surahNo != 9;

  String get revelationArabic => isMakki ? 'مكية' : 'مدنية';

  factory SurahDetailModel.fromAlQuranCloudEditions(
    List<dynamic> rawEditions,
    int surahNo,
  ) {
    final editions = rawEditions
        .map((e) => Map<String, dynamic>.from(e as Map))
        .toList();

    Map<String, dynamic> editionById(String id) {
      return editions.firstWhere(
        (item) {
          final edition = Map<String, dynamic>.from(item['edition'] as Map);
          return edition['identifier'] == id;
        },
        orElse: () => editions.first,
      );
    }

    final quranEdition = editionById(AppConstants.quranTextEdition);

    final editionTexts = <String, Map<int, String>>{};

    for (final surahEdition in editions) {
      final edition = Map<String, dynamic>.from(surahEdition['edition'] as Map);
      final id = edition['identifier'] as String? ?? '';

      final ayahs = (surahEdition['ayahs'] as List? ?? [])
          .map((e) => Map<String, dynamic>.from(e as Map))
          .toList();

      editionTexts[id] = {
        for (final ayah in ayahs)
          (ayah['numberInSurah'] as num).toInt():
              ayah['text'] as String? ?? '',
      };
    }

    final quranAyahs = (quranEdition['ayahs'] as List? ?? [])
        .map((e) => Map<String, dynamic>.from(e as Map))
        .toList();

    final ayahs = quranAyahs.map((ayah) {
      final numberInSurah = (ayah['numberInSurah'] as num).toInt();

      final meanings = <String, String>{};
      for (final entry in AppConstants.meaningEditions.entries) {
        final text = editionTexts[entry.key]?[numberInSurah];
        if (text != null && text.trim().isNotEmpty) {
          meanings[entry.key] = text;
        }
      }

      final tafsirs = <String, String>{};
      for (final entry in AppConstants.tafsirEditionsAr.entries) {
        final text = editionTexts[entry.key]?[numberInSurah];
        if (text != null && text.trim().isNotEmpty) {
          tafsirs[entry.key] = text;
        }
      }

      return AyahModel(
        number: numberInSurah,
        globalNumber: (ayah['number'] as num?)?.toInt() ?? 0,
        page: (ayah['page'] as num?)?.toInt() ?? 0,
        arabicText: ayah['text'] as String? ?? '',
        arabicMeaning: meanings['ar.muyassar'],
        english: meanings['en.sahih'],
        urdu: meanings['ur.junagarhi'],
        bengali: meanings['bn.bengali'],
        meanings: meanings,
        tafsirs: tafsirs,
      );
    }).toList();

    return SurahDetailModel(
      surahNo: surahNo,
      surahName: quranEdition['englishName'] as String? ?? '',
      surahNameArabic: quranEdition['name'] as String? ?? '',
      surahNameArabicLong: quranEdition['name'] as String? ?? '',
      surahNameTranslation:
          quranEdition['englishNameTranslation'] as String? ?? '',
      revelationPlace: quranEdition['revelationType'] as String? ?? '',
      totalAyah: quranEdition['numberOfAyahs'] as int? ?? ayahs.length,
      audio: const {},
      ayahs: ayahs,
    );
  }
}

import 'package:dartz/dartz.dart';
import 'package:dio/dio.dart';

import '../../../../core/api/api_endpoints.dart';
import '../../../../core/constants/app_constants.dart';
import '../../../../core/network/dio_client.dart';
import '../../../../core/network/failures.dart';
import '../models/surah_detail_model.dart';

abstract class ISurahDetailRepository {
  Future<Either<Failure, SurahDetailModel>> getSurah(int surahNo);
}

class SurahDetailRepository implements ISurahDetailRepository {
  final Dio _dio;

  SurahDetailRepository() : _dio = DioClient.instance;

  @override
  Future<Either<Failure, SurahDetailModel>> getSurah(int surahNo) async {
    try {
      final editions = <String>[
        AppConstants.quranTextEdition,
        ...AppConstants.defaultMeaningEditions,
      ];

      final res = await _dio.get(
        ApiEndpoints.surahEditions(surahNo, editions),
      );

      final data = (res.data as Map<String, dynamic>)['data'] as List;

      final model = SurahDetailModel.fromAlQuranCloudEditions(data, surahNo);

      return Right(model);
    } on DioException catch (e) {
      return Left(NetworkFailure(e.message ?? 'خطأ في الشبكة'));
    } catch (e) {
      return Left(UnknownFailure(e.toString()));
    }
  }
}

  Future<Either<Failure, List<SurahInfoModel>>> getSurahList() async {
    try {
      final res = await _dio.get(ApiEndpoints.surahs);

      final data = (res.data as Map<String, dynamic>)['data'] as List;

      final list = data
          .map((e) => SurahInfoModel.fromJson(
                Map<String, dynamic>.from(e as Map),
              ))
          .toList();

      return Right(list);
    } on DioException catch (e) {
      return Left(NetworkFailure(e.message ?? 'خطأ في الشبكة'));
    } catch (e) {
      return Left(UnknownFailure(e.toString()));
    }
  }
}

class _AyahEditionText extends StatelessWidget {
  final int surahNo;
  final int ayahNo;
  final String edition;
  final String? cachedText;
  final TextDirection textDirection;

  const _AyahEditionText({
    required this.surahNo,
    required this.ayahNo,
    required this.edition,
    required this.cachedText,
    required this.textDirection,
  });

  Future<String> _loadText() async {
    if (cachedText != null && cachedText!.trim().isNotEmpty) {
      return cachedText!;
    }

    final res = await DioClient.instance.get(
      ApiEndpoints.ayahEditions(surahNo, ayahNo, [edition]),
    );

    final data = (res.data as Map<String, dynamic>)['data'] as List;

    if (data.isEmpty) return 'لا يوجد محتوى متاح';

    return data.first['text'] as String? ?? 'لا يوجد محتوى متاح';
  }

  @override
  Widget build(BuildContext context) {
    return FutureBuilder<String>(
      key: ValueKey(edition),
      future: _loadText(),
      builder: (context, snapshot) {
        if (snapshot.connectionState == ConnectionState.waiting) {
          return const Center(child: CircularProgressIndicator());
        }

        if (snapshot.hasError) {
          return const Text(
            'حدث خطأ أثناء تحميل المعاني',
            textAlign: TextAlign.center,
          );
        }

        return Directionality(
          textDirection: textDirection,
          child: Text(
            snapshot.data ?? '',
            textAlign: TextAlign.justify,
            style: Theme.of(context).textTheme.bodyLarge?.copyWith(
                  height: 1.9,
                  fontSize: 18,
                ),
          ),
        );
      },
    );

class SurahListScreen extends StatefulWidget {
  const SurahListScreen({super.key});

  @override
  State<SurahListScreen> createState() => _SurahListScreenState();
}

class _SurahListScreenState extends State<SurahListScreen> {
  final SurahListRepository _surahListRepository = SurahListRepository();

  bool _isLoading = true;
  String? _error;
  List<SurahInfoModel> _surahs = [];

  @override
  void initState() {
    super.initState();
    _loadSurahs();
  }

  Future<void> _loadSurahs() async {
    setState(() {
      _isLoading = true;
      _error = null;
    });

    final result = await _surahListRepository.getSurahList();

    result.fold(
          (failure) {
        setState(() {
          _isLoading = false;
          _error = failure.message;
        });
      },
          (surahs) {
        setState(() {
          _isLoading = false;
          _surahs = surahs;
        });
      },
    );
  }

  Future<void> _openSurah(SurahInfoModel surah) async {
    final repo = SurahDetailRepository();

    showDialog(
      context: context,
      barrierDismissible: false,
      builder: (_) => const Center(
        child: CircularProgressIndicator(),
      ),
    );

    final result = await repo.getSurah(surah.surahNo);

    if (!mounted) return;

    Navigator.pop(context);

    result.fold(
          (failure) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text(failure.message)),
        );
      },
          (detail) {
        Navigator.push(
          context,
          MaterialPageRoute(
            builder: (_) => SurahReaderScreen(surah: detail),
          ),
        );
      },
    );
  }

  @override
  Widget build(BuildContext context) {
    return Directionality(
      textDirection: TextDirection.rtl,
      child: Scaffold(
        appBar: AppBar(
          title: const Text('القرآن الكريم'),
          centerTitle: true,
        ),
        body: _buildBody(),
      ),
    );
  }

  Widget _buildBody() {
    if (_isLoading) {
      return const Center(child: CircularProgressIndicator());
    }

    if (_error != null) {
      return Center(
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            Text(_error!),
            const SizedBox(height: 12),
            ElevatedButton(
              onPressed: _loadSurahs,
              child: const Text('إعادة المحاولة'),
            ),
          ],
        ),
      );
    }

    return ListView.builder(
      padding: const EdgeInsets.all(16),
      itemCount: _surahs.length,
      itemBuilder: (context, index) {
        final surah = _surahs[index];

        return SurahCard(
          surah: surah,
          onTap: () => _openSurah(surah),
        );
      },
    );
  }
}

//quran_app1/features/surah_list/surah_helper/quran_text_helper.dart
import 'package:flutter/material.dart';

class QuranTextHelper {
  static const List<String> waqfSigns = [
    'ۘ', 'ۙ', 'ۚ', 'ۛ', 'ۜ', 'ۗ', 'ۖ', 'ە', '۝', '۞',
    'م', 'لا', 'ج', 'صلى', 'قلى',
  ];

  static List<InlineSpan> buildQuranSpans({
    required String text,
    required TextStyle ayahStyle,
    required TextStyle waqfStyle,
  }) {
    final spans = <InlineSpan>[];

    for (int i = 0; i < text.length; i++) {
      final char = text[i];

      if (waqfSigns.contains(char)) {
        spans.add(TextSpan(text: char, style: waqfStyle));
      } else {
        spans.add(TextSpan(text: char, style: ayahStyle));
      }
    }

    return spans;
  }
}

import '../data/models/quran_page_model.dart';
import '../data/models/surah_detail_model.dart';

class SurahPaginationHelper {
  static List<QuranPageModel> paginateAyahs(
      List<AyahModel> ayahs, {
        int ayahsPerPage = 8,
      }) {
    final pages = <QuranPageModel>[];

    for (int i = 0; i < ayahs.length; i += ayahsPerPage) {
      final pageAyahs = ayahs.skip(i).take(ayahsPerPage).toList();

      pages.add(
        QuranPageModel(
          pageIndex: pages.length + 1,
          ayahs: pageAyahs,
        ),
      );
    }

    return pages;
  }
}

class _TafsirText extends StatelessWidget {
  final int surahNo;
  final int ayahNo;
  final String edition;

  const _TafsirText({
    required this.surahNo,
    required this.ayahNo,
    required this.edition,
  });

  Future<String> _loadTafsir() async {
    final res = await DioClient.instance.get(
      ApiEndpoints.ayahEditions(surahNo, ayahNo, [edition]),
    );

    final data = (res.data as Map<String, dynamic>)['data'] as List;

    if (data.isEmpty) return 'لا يوجد تفسير متاح لهذه الآية';

    return data.first['text'] as String? ?? 'لا يوجد تفسير متاح لهذه الآية';
  }

  @override
  Widget build(BuildContext context) {
    return FutureBuilder<String>(
      key: ValueKey(edition),
      future: _loadTafsir(),
      builder: (context, snapshot) {
        if (snapshot.connectionState == ConnectionState.waiting) {
          return const Center(child: CircularProgressIndicator());
        }

        if (snapshot.hasError) {
          return const Text(
            'حدث خطأ أثناء تحميل التفسير',
            textAlign: TextAlign.center,
          );
        }

        return Text(
          snapshot.data ?? '',
          textAlign: TextAlign.justify,
          style: Theme.of(context).textTheme.bodyLarge?.copyWith(
                height: 1.9,
                fontSize: 18,
              ),
        );
      },
    );
  }
}

// lib/features/tafsir/data/models/tafsir_model.dart

class TafsirEntry {
  final String author;
  final String? groupVerse;
  final String content;

  const TafsirEntry({
    required this.author,
    this.groupVerse,
    required this.content,
  });

  factory TafsirEntry.fromJson(Map<String, dynamic> json) => TafsirEntry(
    author: json['author'] as String? ?? '',
    groupVerse: json['groupVerse'] as String?,
    content: json['content'] as String? ?? '',
  );
}

class AyahTafsirModel {
  final String surahName;
  final int surahNo;
  final int ayahNo;
  final List<TafsirEntry> tafsirs;

  const AyahTafsirModel({
    required this.surahName,
    required this.surahNo,
    required this.ayahNo,
    required this.tafsirs,
  });

  factory AyahTafsirModel.fromJson(Map<String, dynamic> json) {
    final rawTafsirs = (json['tafsirs'] as List?) ?? [];
    return AyahTafsirModel(
      surahName: json['surahName'] as String? ?? '',
      surahNo:   json['surahNo']   as int?    ?? 0,
      ayahNo:    json['ayahNo']    as int?    ?? 0,
      tafsirs: rawTafsirs
          .map((e) => TafsirEntry.fromJson(e as Map<String, dynamic>))
          .toList(),
    );
  }
}
